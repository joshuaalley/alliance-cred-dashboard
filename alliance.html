Preliminary model output — not for citation. Each line is a fused human + LLM IRT estimate of alliance credibility (latent α; higher = more credible). Shaded bands are 90% / 50% credible intervals. Tiers reflect expert coverage, not precision. Vintage: 2026-07-03.

estimates = FileAttachment("data/estimates.csv").csv({typed: true})
meta = FileAttachment("data/alliance-metadata.json").json()
// Deep-linkable: ?atopid=3220 pre-selects that alliance (the home table links in this way);
// falls back to the first alliance alphabetically.
viewof allianceId = {
  const opts = meta.slice().sort((a, b) => a.name.localeCompare(b.name));
  const q = +(new URLSearchParams(location.search).get("atopid"));
  const initial = opts.some(d => d.atopid === q) ? q : opts[0].atopid;
  return Inputs.select(
    new Map(opts.map(d => [`${d.name} (${d.begyr}–${d.endyr})`, d.atopid])),
    {label: "Alliance", value: initial}
  );
}
am   = meta.find(d => d.atopid === allianceId)
rows = estimates.filter(d => d.alliance_id === allianceId)

// Empty recipient (aggregate actors like "partners"/"Members") -> a readable label.
recipLabel = r => (r === null || r === undefined || `${r}`.length === 0) ? "(aggregate)" : `${r}`

// Option lists derived from the DATA (the series that actually exist), each with "All".
sources = ["All", ...Array.from(new Set(rows.map(d => d.cred_source)))
                       .sort((a, b) => `${a}`.localeCompare(`${b}`))]
recips  = ["All", ...Array.from(new Set(rows.map(d => recipLabel(d.cred_recip))))
                       .sort((a, b) => a.localeCompare(b))]
viewof source = Inputs.select(sources, {label: "Source (whose credibility)"})
viewof recip = Inputs.select(recips, {label: "Toward (recipient)"})
shown = rows.filter(d =>
  (source === "All" || d.cred_source === source) &&
  (recip  === "All" || recipLabel(d.cred_recip) === recip))

seriesOf = d => d.cred_source +
  (recipLabel(d.cred_recip) === "(aggregate)" ? "" : " → " + d.cred_recip)

nSeries = new Set(shown.map(seriesOf)).size
focused = nSeries === 1
Plot.plot({
  width: 920, height: 470, marginLeft: 60, marginBottom: 40,
  x: {label: "Year", tickFormat: "d"},
  y: {label: "↑ latent credibility (α)", grid: true},
  color: {legend: nSeries > 1},
  marks: [
    // Credible-interval ribbons only when one series is focused (overlapping bands are unreadable).
    focused ? Plot.areaY(shown, {x: "year", y1: "irt_q5",  y2: "irt_q95", fill: "#1f5fa8", fillOpacity: 0.12}) : null,
    focused ? Plot.areaY(shown, {x: "year", y1: "irt_q25", y2: "irt_q75", fill: "#1f5fa8", fillOpacity: 0.22}) : null,
    Plot.line(shown, {x: "year", y: "irt_median", stroke: seriesOf, strokeWidth: 1.8}),
    Plot.ruleY([0], {strokeOpacity: 0.15})
  ].filter(m => m)
})
html`<div style="font-size:0.92rem; color:#333; max-width:920px">

  <b>${["gold", "silver", "bronze"].find(t => new Set(shown.map(d => d.confidence_tier)).has(t)) ?? "—"}</b>.
  ${new Set(shown.map(d => d.confidence_tier)).has("bronze")

    : ""}
  <br><br>${am.intro_para}
</div>`
